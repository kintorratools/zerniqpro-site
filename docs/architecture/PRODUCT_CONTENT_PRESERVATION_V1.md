## Appendix: Content Collection Schema Reference

Based on actual usage in `ProductDetail.astro`, the effective schema is:

```typescript
interface ProductEntry {
  title: string;
  description: string;
  main: {
    id: number; // 1..4, determines sort order
    content: string; // marketing blurb
    imgCard: string; // @/images/product-image-{id}.avif
    imgMain: string; // @/images/product-image-main-{id}.avif
    imgAlt: string; // alt text for images
  };
  tabs: Array<{
    id: string; // tab element id, e.g. "tabs-with-card-item-1"
    dataTab: string; // target selector, e.g. "#tabs-with-card-1"
    title: string; // tab label
  }>;
  longDescription: {
    title: string;
    subTitle: string;
    btnTitle: string; // CTA button text
    btnURL: string; // CTA button link
  };
  descriptionList: Array<{
    title: string;
    subTitle: string;
  }>;
  specificationsLeft: Array<{
    title: string;
    subTitle: string;
  }>;
  specificationsRight?: Array<{
    // OPTIONAL — omitted when tableData present
    title: string;
    subTitle: string;
  }>;
  tableData?: Array<{
    // OPTIONAL — only on item-a765
    feature: [string, string]; // header row: [col1, col2]
    description: Array<[string, string]>; // data rows
  }>;
  blueprints: {
    first: string; // @/images/blueprint-1.avif
    second: string; // @/images/blueprint-2.avif
  };
}
```
